// A deliberately different language, to show what "flexible" buys you.
//
// Differences from grammar.txt:
//   - new keywords declared here, not in the C source (let, loop, unless)
//   - a new punctuation token (:=) declared here
//   - no parentheses around if/loop conditions
//   - a rule name the C enum has never heard of (STX_PROGRAM), minted on sight
//   - a different start symbol (the first rule in the file)
//
// The binary is unchanged. Run with:
//   ./bin/main.run -g ./resources/grammar-mini.txt -s ./resources/mini.tl

#keyword TK_LET    "let" ;
#keyword TK_LOOP   "loop" ;
#keyword TK_UNLESS "unless" ;
#token   TK_ASSIGN ":=" ;

// STX_PROGRAM is not in syntax.def. It is interned at load time and treated as
// a transparent grouping at run time.
STX_PROGRAM = STX_SCOPE ;

STX_SCOPE = TK_LBRACE, {STX_INIT | STX_ASSIGN | STX_IF | STX_WHILE | STX_ECHO}, TK_RBRACE ;

// note the reshaped rules: the variable is no longer the second element
STX_INIT   = TK_LET, STX_VAR, TK_ASSIGN, STX_EXPR, TK_SEMICOLON ;
STX_ASSIGN = STX_VAR, STX_ASSIGNOP, STX_EXPR, TK_SEMICOLON ;
STX_ECHO   = (STX_FNCALL | STX_EXPR), TK_SEMICOLON ;

// conditions without parentheses, and an inverted conditional
STX_IF    = TK_IF, STX_EXPR, STX_SCOPE, [TK_ELSE, STX_SCOPE] ;
STX_WHILE = TK_LOOP, STX_EXPR, STX_SCOPE ;

STX_EXPR = [STX_NOT | STX_NEGATE], (STX_GEXPR | STX_FNCALL | STX_VAR | STX_NUM | STX_STRLIT | STX_TRUE | STX_FALSE), {(STX_ARITHOP | STX_BOOLOP), [STX_NOT | STX_NEGATE], (STX_GEXPR | STX_FNCALL | STX_VAR | STX_NUM | STX_STRLIT | STX_TRUE | STX_FALSE)} ;

STX_GEXPR  = TK_LPAREN, STX_EXPR, TK_RPAREN ;
STX_FNCALL = TK_IDENTIFIER, TK_LPAREN, [STX_EXPR], {TK_COMMA, STX_EXPR}, TK_RPAREN ;

STX_VAR    = TK_IDENTIFIER ;
STX_NUM    = TK_NUM ;
STX_STRLIT = TK_CHARS ;

STX_ARITHOP  = STX_MULT | STX_DIV | STX_SUM | STX_DIFF | STX_MOD ;
STX_BOOLOP   = STX_GREATER | STX_LESS | STX_NOT_EQUAL | STX_EQUAL_EQUAL | STX_GREATER_EQUAL | STX_LESS_EQUAL | STX_AND | STX_OR ;
STX_ASSIGNOP = STX_EQUAL | STX_PLUS_EQUAL | STX_MINUS_EQUAL ;

STX_MULT = TK_STAR ;
STX_DIV  = TK_FSLASH ;
STX_SUM  = TK_PLUS ;
STX_DIFF = TK_MINUS ;
STX_MOD  = TK_MOD ;

STX_TRUE          = TK_TRUE ;
STX_FALSE         = TK_FALSE ;
STX_AND           = TK_AND ;
STX_OR            = TK_OR ;
STX_NOT           = TK_BANG ;
STX_GREATER       = TK_GREATER ;
STX_LESS          = TK_LESS ;
STX_EQUAL_EQUAL   = TK_EQUAL_EQUAL ;
STX_NOT_EQUAL     = TK_BANG_EQUAL ;
STX_GREATER_EQUAL = TK_GREATER_EQUAL ;
STX_LESS_EQUAL    = TK_LESS_EQUAL ;

STX_EQUAL       = TK_EQUAL ;
STX_PLUS_EQUAL  = TK_PLUS, TK_EQUAL ;
STX_MINUS_EQUAL = TK_MINUS, TK_EQUAL ;
STX_NEGATE      = TK_MINUS ;
